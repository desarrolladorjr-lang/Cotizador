# Módulo CXC/CXP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Agregar al dashboard CRM un módulo de solo lectura "CXC/CXP" que muestra facturas con saldo pendiente de cobro y de pago, leídas del spreadsheet externo EXPORTACIONES 2026, agrupadas por cliente (CXC) u origin (CXP), con subtotales y total por segmento.

**Architecture:** Nuevo cliente Sheets (`cxcCxpClient.ts`) sigue el patrón de spreadsheet externo de `ordenesClient.ts`, reutilizando `getSheetsApi()`. Un módulo de schema (`cxcCxpSchema.ts`) mapea índices de columna reales por hoja/tipo. Un normalizador (`normalizeCxcCxp.ts`) filtra filas con saldo pendiente y las agrupa. Un endpoint (`GET /api/cxc-cxp`) sin cache propio expone el resultado. Un componente cliente (`CxcCxpPanel.tsx`) auto-fetch con poll de 15 min (solo mientras está montado/activo) + botón manual, renderiza sub-tabs Marítimo/Terrestre con tablas agrupadas. Se integra en `page.tsx` como nueva sección del sidebar existente.

**Tech Stack:** Next.js 14 App Router, TypeScript, `googleapis`, Vitest + Testing Library (mismo stack que el resto de `./dashboard`).

## Global Constraints

- Spreadsheet ID: `14ep3kX8urvZlwHwcdMJZEf6V6aIJWs1ZWxCyNVb-uxc` (EXPORTACIONES 2026), hojas `MARITIMOS` y `TERRESTRES` — distinto al `SPREADSHEET_ID` del CRM principal.
- Solo lectura: no se agrega ningún endpoint de escritura ni se toca `sheetsWriter.ts`.
- Filtro CXC: `INVOICE` no vacío Y `AR` ≠ 0/vacío.
- Filtro CXP: `INV SUP` no vacío Y `AMOUNT SUP` ≠ `$ TO CASH` (tratando vacío como 0).
- Filas de datos empiezan en la fila 3 del sheet (índice 2 del array `rows`, ya que la fila 1 es un título de sección y la fila 2 es el encabezado real de columnas).
- Sin cache propio en el endpoint (fetch fresco por request); el control de frecuencia lo hace el poll de 15 min del cliente + botón manual.
- Estilo visual: reutiliza las variables CSS existentes (`--orange`, `--orange-soft`, `--cream`, `--cream-dim`, `--line`, `--grey`, `--font-mono`, `--font-display`) — no se introduce paleta nueva.
- Todos los archivos nuevos van en `dashboard/src/lib/` y `dashboard/src/components/` y `dashboard/src/app/api/cxc-cxp/`, siguiendo la estructura ya usada por `ordenesClient.ts` / `/api/ordenes/route.ts` / `OrdenesCompra.tsx`.

// Spreadsheet EXPORTACIONES 2026: fuente de CXC/CXP, distinto al spreadsheet del CRM.
// Requiere compartir como Viewer con el client_email del service account.
export const CXC_CXP_SPREADSHEET_ID = '14ep3kX8urvZlwHwcdMJZEf6V6aIJWs1ZWxCyNVb-uxc';

export type CxcCxpSheetName = 'MARITIMOS' | 'TERRESTRES';

const RANGES: Record<CxcCxpSheetName, string> = {
  MARITIMOS: 'MARITIMOS!A1:BA2000',
  TERRESTRES: 'TERRESTRES!A1:AW2000',
};

export interface CxcCxpFetchResult {
  sheetName: CxcCxpSheetName;
  rows: string[][] | null;
  error?: string;
}

async function fetchSheetRows(sheetName: CxcCxpSheetName, api: SheetsApi): Promise<string[][]> {
  const res = await api.spreadsheets.values.get({
    spreadsheetId: CXC_CXP_SPREADSHEET_ID,
    range: RANGES[sheetName],
  });
  return (res.data.values as string[][]) || [];
}

export async function fetchAllCxcCxpSheets(sheetsApi?: SheetsApi): Promise<CxcCxpFetchResult[]> {
  const api = sheetsApi || (await getSheetsApi());
  return Promise.all(
    (['MARITIMOS', 'TERRESTRES'] as CxcCxpSheetName[]).map(async (sheetName): Promise<CxcCxpFetchResult> => {
      try {
        const rows = await fetchSheetRows(sheetName, api);
        return { sheetName, rows };
      } catch (err) {
        return { sheetName, rows: null, error: (err as Error).message };
      }
    })
  );
}
```

export interface CxcFieldIndex {
  client: number;
  seller: number;
  contract: number;
  material: number;
  invoice: number;
  date: number;
  amount: number;
  toCash: number;
  ar: number;
}

export interface CxpFieldIndex {
  client: number;
  material: number;
  origin: number;
  supplier: number;
  invSup: number;
  date: number;
  kgSup: number;
  pct: number;
  dlr: number;
  priceSup: number;
  amountSup: number;
  toCash: number;
  paymntDate: number;
  disctToSup: number; // -1 si la hoja no tiene esta columna (usa `deducted` en su lugar)
  gprofit: number;
}

export const CXC_SCHEMA: Record<Segmento, CxcFieldIndex> = {
  maritimo: {

    invoice: 18, date: 19, amount: 23, toCash: 24, ar: 30,

  terrestre: {
    client: 0, seller: 1, contract: 3, material: 4,
    invoice: 15, date: 16, amount: 20, toCash: 21, ar: 28,
  },

export const CXP_SCHEMA: Record<Segmento, CxpFieldIndex> = {
  maritimo: {
    client: 0, material: 4,
    origin: 40, supplier: 41, invSup: 42, date: 43, kgSup: 44, pct: 45, dlr: 46,
    priceSup: 47, amountSup: 48, toCash: 49, paymntDate: 51, disctToSup: -1, gprofit: 52,
  },
  terrestre: {
    client: 0, material: 4,
    origin: 36, supplier: 37, invSup: 38, date: 39, kgSup: 40, pct: 41, dlr: 42,
    priceSup: 43, amountSup: 44, toCash: 45, paymntDate: 46, disctToSup: 47, gprofit: 48,
  },
};

// Índice negativo (columna ausente en esa hoja) → celda vacía en vez de reventar.
export function getCell(row: string[], index: number): string {
  if (index < 0) return '';
  return (row[index] ?? '').toString().trim();
}

import { CXC_SCHEMA, type Segmento, getCell } from './cxcCxpSchema';
import { parseNumero } from './ordenesClient';

export interface CxcRow {
  seller: string;
  contract: string;
  material: string;
  invoice: string;
  date: string;
  amount: number;
  toCash: number;
  ar: number;
}

export interface CxcGroup {
  cliente: string;
  rows: CxcRow[];
  subtotal: number;
}

export interface CxcTable {
  grupos: CxcGroup[];
  total: number;
}

// Filas de datos empiezan en la fila 3 del sheet (fila 1 = título de sección, fila 2 = encabezado real).
const DATA_START_ROW = 2;

export function buildCxcTable(rows: string[][], segmento: Segmento): CxcTable {
  const schema = CXC_SCHEMA[segmento];
  const dataRows = rows.slice(DATA_START_ROW);

  const groupOrder: string[] = [];
  const groups = new Map<string, CxcRow[]>();

  for (const row of dataRows) {
    if (!row) continue;
    const invoice = getCell(row, schema.invoice);
    const ar = parseNumero(getCell(row, schema.ar));
    if (!invoice || !ar) continue; // ar null o 0 → sin saldo pendiente

    const cliente = getCell(row, schema.client) || '(sin cliente)';
    const cxcRow: CxcRow = {
      seller: getCell(row, schema.seller),
      contract: getCell(row, schema.contract),
      material: getCell(row, schema.material),
      invoice,
      date: getCell(row, schema.date),
      amount: parseNumero(getCell(row, schema.amount)) ?? 0,
      toCash: parseNumero(getCell(row, schema.toCash)) ?? 0,
      ar,
    };

    if (!groups.has(cliente)) {
      groups.set(cliente, []);
      groupOrder.push(cliente);
    }
    groups.get(cliente)!.push(cxcRow);
  }

  const grupos: CxcGroup[] = groupOrder.map((cliente) => {
    const groupRows = groups.get(cliente)!;
    const subtotal = groupRows.reduce((sum, r) => sum + r.ar, 0);
    return { cliente, rows: groupRows, subtotal };
  });
  const total = grupos.reduce((sum, g) => sum + g.subtotal, 0);

  return { grupos, total };
}
```

  cliente: string;
  material: string;
  origin: string;
  supplier: string;
  invSup: string;
  date: string;
  kgSup: number;
  pct: string;
  dlr: string;

  toCash: number;
  paymntDate: string;
  disctToSup: number | null; // null si la hoja no tiene esta columna

export interface CxpGroup {

  grupos: CxpGroup[];
  total: number;

export function buildCxpTable(rows: string[][], segmento: Segmento): CxpTable {
  const schema = CXP_SCHEMA[segmento];
  const dataRows = rows.slice(DATA_START_ROW);

  const groupOrder: string[] = [];

  for (const row of dataRows) {
    if (!row) continue;
    const invSup = getCell(row, schema.invSup);
    if (!invSup) continue;

    const amountSup = parseNumero(getCell(row, schema.amountSup)) ?? 0;
    const toCash = parseNumero(getCell(row, schema.toCash)) ?? 0;
    if (amountSup === toCash) continue; // sin saldo pendiente

    const origin = getCell(row, schema.origin) || '(sin origin)';
    const cxpRow: CxpRow = {
      cliente: getCell(row, schema.client),
      material: getCell(row, schema.material),
      origin,
      supplier: getCell(row, schema.supplier),
      invSup,
      date: getCell(row, schema.date),
      kgSup: parseNumero(getCell(row, schema.kgSup)) ?? 0,
      pct: getCell(row, schema.pct),
      dlr: getCell(row, schema.dlr),
      priceSup: parseNumero(getCell(row, schema.priceSup)) ?? 0,
      amountSup,
      toCash,
      paymntDate: getCell(row, schema.paymntDate),
      disctToSup: parseNumero(getCell(row, schema.disctToSup)),
      gprofit: parseNumero(getCell(row, schema.gprofit)) ?? 0,
      saldoPendiente: amountSup - toCash,
    };

    if (!groups.has(origin)) {
      groups.set(origin, []);
      groupOrder.push(origin);
    }
    groups.get(origin)!.push(cxpRow);
  }

  const grupos: CxpGroup[] = groupOrder.map((origin) => {
    const groupRows = groups.get(origin)!;
    const subtotal = groupRows.reduce((sum, r) => sum + r.saldoPendiente, 0);
    return { origin, rows: groupRows, subtotal };
  });
  const total = grupos.reduce((sum, g) => sum + g.subtotal, 0);

  return { grupos, total };
}

import { fetchAllCxcCxpSheets } from '@/lib/cxcCxpClient';
import { buildCxcTable, buildCxpTable, type CxcTable, type CxpTable } from '@/lib/normalizeCxcCxp';

// Lee el spreadsheet EXPORTACIONES 2026 por request — sin cache propio, el
// cliente controla la frecuencia (poll 15 min + botón manual).
export const dynamic = 'force-dynamic';

interface SegmentoResponse {
  cxc: CxcTable;
  cxp: CxpTable;
  error?: string;
}

const EMPTY: SegmentoResponse = { cxc: { grupos: [], total: 0 }, cxp: { grupos: [], total: 0 } };

export async function GET() {
  const results = await fetchAllCxcCxpSheets();
  const maritimoResult = results.find((r) => r.sheetName === 'MARITIMOS');
  const terrestreResult = results.find((r) => r.sheetName === 'TERRESTRES');

  const maritimo: SegmentoResponse = maritimoResult?.rows
    ? { cxc: buildCxcTable(maritimoResult.rows, 'maritimo'), cxp: buildCxpTable(maritimoResult.rows, 'maritimo') }
    : { ...EMPTY, error: maritimoResult?.error || 'hoja MARITIMOS no disponible' };

  const terrestre: SegmentoResponse = terrestreResult?.rows
    ? { cxc: buildCxcTable(terrestreResult.rows, 'terrestre'), cxp: buildCxpTable(terrestreResult.rows, 'terrestre') }
    : { ...EMPTY, error: terrestreResult?.error || 'hoja TERRESTRES no disponible' };

  return Response.json({ maritimo, terrestre, lastUpdated: new Date().toISOString() });
}
